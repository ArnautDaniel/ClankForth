cmake_minimum_required(VERSION 3.20)
project(clankforth VERSION 0.1.0 LANGUAGES C)

# A project of its own, or inside a host (add_subdirectory): its tests by default only on its own
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(CLANKFORTH_TOP_LEVEL ON)
    if(NOT CMAKE_BUILD_TYPE)
        set(CMAKE_BUILD_TYPE Debug)
    endif()
else()
    set(CLANKFORTH_TOP_LEVEL OFF)
endif()
option(CLANKFORTH_TESTS "the language's tests (ctest)" ${CLANKFORTH_TOP_LEVEL})

include(cmake/embed.cmake)
file(GLOB CLANKFORTH_LIB_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/lib/*.fs")
list(SORT CLANKFORTH_LIB_SOURCES)
clankforth_embed("${CMAKE_CURRENT_BINARY_DIR}/forth_sources.c" ${CLANKFORTH_LIB_SOURCES})

# the Forth system: forth.h, the actor kernel (actors.h), and the tool's main for hosts' own tools (forth_tool_main)
add_library(clankforth STATIC src/forth.c src/actors.c src/tool.c "${CMAKE_CURRENT_BINARY_DIR}/forth_sources.c")
target_include_directories(clankforth PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(clankforth PUBLIC m)
set_target_properties(clankforth PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS ON)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(clankforth PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
# (this repo's own builds - its CI - stop at a warning; a host's compiler may know warnings ours doesn't)
option(CLANKFORTH_WERROR "warnings are errors" OFF)
if(CLANKFORTH_WERROR AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(clankforth PRIVATE -Werror)
endif()
# optimized in a debug build too (asserts and the lent-memory checks stay: NDEBUG unset) - loading a host's scripts
# is mostly the compiler's lookups, many times slower unoptimized. Off: CLANKFORTH_OPTIMIZE=OFF, to step through it.
option(CLANKFORTH_OPTIMIZE "optimize the Forth system in a debug build" ON)
if(CLANKFORTH_OPTIMIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(clankforth PRIVATE $<$<OR:$<CONFIG:Debug>,$<CONFIG:>>:-O2>)
endif()

add_executable(clankforth-tool src/main.c)
set_target_properties(clankforth-tool PROPERTIES OUTPUT_NAME clankforth)
target_link_libraries(clankforth-tool PRIVATE clankforth)

if(CLANKFORTH_TESTS)
    enable_testing()
    file(GLOB CLANKFORTH_TEST_FILES "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.fs")
    foreach(t ${CLANKFORTH_TEST_FILES})
        get_filename_component(name "${t}" NAME_WE)
        add_test(NAME ${name} COMMAND clankforth-tool "${t}")
    endforeach()
    add_executable(host_call tests/host_call.c)   # (the host's side, from C: forth_call from a primitive)
    target_link_libraries(host_call PRIVATE clankforth)
    add_test(NAME host_call COMMAND host_call)
endif()

# a fuzzer over forth_eval (tests/fuzz_eval.c): clang's libFuzzer, with the sanitizers
option(CLANKFORTH_FUZZ "the fuzzer (clang: libFuzzer)" OFF)
if(CLANKFORTH_FUZZ)
    target_compile_options(clankforth PUBLIC -fsanitize=fuzzer-no-link,address,undefined)
    target_link_options(clankforth PUBLIC -fsanitize=address,undefined)
    add_executable(fuzz_eval tests/fuzz_eval.c)
    target_link_libraries(fuzz_eval PRIVATE clankforth)
    target_link_options(fuzz_eval PRIVATE -fsanitize=fuzzer)
endif()
